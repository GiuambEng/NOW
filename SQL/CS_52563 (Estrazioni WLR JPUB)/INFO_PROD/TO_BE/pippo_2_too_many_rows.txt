CREATE OR REPLACE FUNCTION PIPPO_BIS (P_BDS_ID IN BDS_1_CASEDATA_ORDINE.BDS_ID%TYPE,
                                      P_IDRICHIESTAOLO IN BDS_1_CASEDATA_ORDINE.IDRICHIESTAOLO%TYPE,
                                      P_TIPO_LINEA IN VARCHAR2)
  RETURN NUMBER
  DETERMINISTIC
  PARALLEL_ENABLE
IS
 V_ATTRIBUTE_VALUE   VARCHAR2(100);
BEGIN
-- CS_YYY INIZIO
-- ATTRIBUTE_VALUE := CODIFICA_ATTRIBUTO_ORDINE(BDS_ID, 'CODICEFATTURAZIONE');
BEGIN
   SELECT ATTR.VALOREATTRIBUTO INTO V_ATTRIBUTE_VALUE FROM
      APPUSER.BDS_1_CASEDATA_ORDINE ORD,
      APPUSER.BDS_1_CASEDATA_DETTAGLIOORDINE COMP,
      APPUSER.BDS_1_CASEDATA_ATTRIBTDTTGLRDN ATTR
   WHERE ORD.BDS_ID=COMP.ORDINE_DETTAGLIOORDINE_BDSID
      AND COMP.BDS_ID=ATTR.DTTGLRDN_TTRBTDTTGLIRDIN_BDSID
      AND ORD.BDS_ID= P_BDS_ID
      AND ATTR.NOMEATTRIBUTO='CODICEFATTURAZIONE'
      AND COMP.NOMECOMPONENTE='ACCESSOWLR';
EXCEPTION
   WHEN TOO_MANY_ROWS THEN
      V_ATTRIBUTE_VALUE := 'Z1';
   WHEN OTHERS THEN
      NULL;
END;

IF V_ATTRIBUTE_VALUE IS NULL THEN
   BEGIN
      SELECT ATTR.VALOREATTRIBUTO INTO V_ATTRIBUTE_VALUE FROM
         APPUSER.BDS_1_CASEDATA_ASSET AST,
         APPUSER.BDS_1_CASEDATA_DETTAGLIOASSET COMP,
         APPUSER.BDS_1_CASEDATA_ATTRIBTDTTGLSST ATTR
      WHERE AST.BDS_ID=COMP.ASSET_DETTAGLIOASSET_BDSID
         AND COMP.BDS_ID=ATTR.DTTGLSST_TTRBTIDTTGLISST_BDSID
         AND ATTR.NOMEATTRIBUTO='CODICEFATTURAZIONE'
         AND AST.IDRICHIESTAOLO = P_IDRICHIESTAOLO
         AND COMP.NOMECOMPONENTE='ACCESSOWLR';
   EXCEPTION
      WHEN TOO_MANY_ROWS THEN
         V_ATTRIBUTE_VALUE := 'Z2';
      WHEN OTHERS THEN
         NULL;
   END;
END IF;
-- CS_YYY FINE

IF P_TIPO_LINEA ='RES' THEN
  CASE
     WHEN V_ATTRIBUTE_VALUE = 'Z1' THEN RETURN 91;
     WHEN V_ATTRIBUTE_VALUE = 'Z2' THEN RETURN 92;
     WHEN V_ATTRIBUTE_VALUE IN ('A','B','C','U') THEN RETURN 50;
     WHEN V_ATTRIBUTE_VALUE IN ('D','E','F','G','H','I','L','M','N','O','P','Q','R','S','T') THEN RETURN 51;
      ELSE RETURN NULL;
  END CASE;

ELSIF P_TIPO_LINEA ='DIR' THEN
   CASE
     WHEN V_ATTRIBUTE_VALUE = 'Z1' THEN RETURN 91;
     WHEN V_ATTRIBUTE_VALUE = 'Z2' THEN RETURN 92;
     WHEN V_ATTRIBUTE_VALUE IN ('C','E','H','M') THEN RETURN 51;
     WHEN V_ATTRIBUTE_VALUE IN ('A','B','D','F','I','L','N','O','P','Q','R','S','T','U') THEN RETURN 52;
     WHEN V_ATTRIBUTE_VALUE IN ('G') THEN RETURN 53;
      ELSE RETURN NULL;
   END CASE;

ELSIF P_TIPO_LINEA ='TIPOLINEA' THEN
   CASE
     WHEN V_ATTRIBUTE_VALUE = 'Z1' THEN RETURN 91;
     WHEN V_ATTRIBUTE_VALUE = 'Z2' THEN RETURN 92;
     WHEN V_ATTRIBUTE_VALUE IN ('A','C','D','E','U') THEN RETURN 55;
     WHEN V_ATTRIBUTE_VALUE IN ('B','F','G','H','I','L') THEN RETURN 53;
     WHEN V_ATTRIBUTE_VALUE IN ('M','N','Q','P') THEN RETURN 54;
      ELSE RETURN NULL;
   END CASE;
END IF;

END PIPPO_BIS;
